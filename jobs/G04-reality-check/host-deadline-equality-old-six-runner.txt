async function snapshot(page:Page):Promise<Snapshot>{
 const observed=await page.evaluate(()=>{
  const publicEl=document.getElementById('public')!,privateEl=document.getElementById('private')!;
  return {dateNow:Date.now(),performanceNow:performance.now(),phase:publicEl.getAttribute('data-phase'),publicText:publicEl.textContent??'',privateText:privateEl.textContent??'',privateOpen:privateEl.getAttribute('data-open'),privateInputCount:privateEl.querySelectorAll('input,textarea').length,clock:document.getElementById('countdown')?.textContent??null,controllerClock:document.getElementById('controller-countdown')?.textContent??null,scores:Array.from(document.querySelectorAll('.score strong'),e=>Number(e.textContent)),menu:Array.from(document.querySelectorAll<HTMLButtonElement>('[data-choice]'),e=>({id:e.dataset.choice!,text:e.textContent??'',disabled:e.disabled})),skipDisabled:(document.getElementById('skip') as HTMLButtonElement).disabled};
 });
 activeCase?.snapshots.push(observed);
 return observed;
}
function roster(bot=false){return [0,1].map(i=>({id:`p${i}`,name:`Player ${i+1}`,avatarId:'🙂',connected:true,bot:bot&&i===1}));}
function findSeed(mode:'quick'|'bluff',kind?:string):number{
 for(let seed=1;seed<1000;seed++){
  const state=game.init({players:roster(),seed,now:101000,settings:{mode,rounds:4}});
  if(!kind||state.question.kind===kind)return seed;
 }
 throw new Error('No fixture seed for '+mode+'/'+kind);
}

 let state=game.init({players:roster(bot),seed,now:await page.evaluate(()=>Date.now()),settings:{mode,rounds:4}});
 const steps:Step[]=[];
 async function click(selector:string,event?:ReplayEvent):Promise<number>{
  const before=await snapshot(page),now=before.dateNow;
  await page.click(selector);
  if(event)state=game.reduce(state,{...event,now});
  steps.push({selector,now,event,before,after:await snapshot(page)});
  return now;
 }
 await page.click('#start');await click('#skip',{type:'vip',action:'skip'});await click('#skip',{type:'vip',action:'skip'});
 return {context,page,seed,steps,get state(){return state;},click};
}
async function jumpPast(f:Fixture,deadline:number,advanceMs=100):Promise<Snapshot>{
 await f.page.clock.setSystemTime(deadline+10);
 if(advanceMs)await f.page.clock.runFor(advanceMs);
 await new Promise<void>(resolve=>setTimeout(resolve,150));
 return snapshot(f.page);
}
const deadlineOf=(state:State):number=>{if(state.phase.deadline===null)throw new Error('Fixture has no deadline');return state.phase.deadline;};
const timerFirst=(state:State,now:number):State=>game.reduce(state,{type:'timer',phaseId:state.phase.id,startedAt:state.phase.startedAt,now});
const expectedScores=(state:State)=>game.tvView(state).players.map(p=>p.score??0);
const matchTimer=(actual:Snapshot,expected:State)=>{assert.equal(actual.phase,expected.phase.id,'expired timer must advance before the input');assert.deepEqual(actual.scores,expectedScores(expected));};
async function runCase<T>(name:string,gather:()=>Promise<T>,validate:(evidence:T)=>void):Promise<void>{
 const item:CaseReport={name,status:'pending',snapshots:[]};activeCase=item;report.cases.push(item);save();
 try{const evidence=await gather();item.evidence=evidence;save();validate(evidence);item.status='passed';}
 catch(error){item.status='failed';item.error=error instanceof Error?error.stack??error.message:String(error);report.failures.push(name+': '+item.error);console.error(item.error);}
 save();activeCase=null;
}
